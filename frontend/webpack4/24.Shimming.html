# ProvidePlugin

new webpack.ProvidePlugin({
    // 发现代码中使用了$，则会自动在代码引入jquery模块
    $: 'jquery',
    _: 'lodash',
    // 在代码中使用_map，相当于使用lodash的map方法
    _map: ['lodash', 'map']
})
